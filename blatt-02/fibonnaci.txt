# Code is written with goal not to use IN2 because this could be used as stack-Pointer
LOADI 6
STORE 30            # Load n in Memory

# Check for edge Cases (n==1 or n==2) and Prepare Memory
LOAD 30
JUMPC = 17        # Jump if n <= 0
SUBI 2
JUMPC <= 18    # Jump if n <= 2
STORE 30
LOADI 1
STORE 0
MOVE acc in1        # Store f(0) = f(1) = 1 in M(0) and IN1

# f(n-1) is stored in IN1, f(n-2) is stored in M(0)
# To copy f(n-1) to M(0) after calculating f(n) while not using IN2,
# M(0) = ACC - M(0) = f(n) - f(n-1)

MOVE IN1 ACC
ADD 0               # f(n-1) - f(n-2)
MOVE ACC IN1
SUB 0               # f(n) - f(n-1) = f(n-2)
STORE 0             # Update f(n-2) in M(0)
LOAD 30
SUBI 1
JUMPC = 9
STORE 30
JUMP -9

# if n == 0, return 0
LOADI 0
STORE 33
JUMP 7

# if n == 1 or n == 2, return 1
LOADI 1
STORE 33
JUMP 4

# if done, result is in IN1
MOVE IN1 ACC
STORE 33
JUMP 1

PRINT 33
JUMP 0
