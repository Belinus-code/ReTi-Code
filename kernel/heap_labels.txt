JUMP PREPARE_STACK
SETUP:

# --- SETUP ---
LOADI 4095
STORE 4095             # Initialisiere leeren Heap

# --- PHASE 1: Drei Blöcke allokieren ---
# Block A (n=10)
LOADI 1
STORE 1026
LOADI 10
STORE 1030
LOADI *CALLOC
STORE 1024
MOVE pc acc
STORE 1025
JUMP CALL
LOAD 1030
STORE 10               # M(10) = 4098 (Pointer A)
PRINT 10

# Block B (n=10)
LOADI 1
STORE 1026
LOADI 10
STORE 1030
LOADI *CALLOC
STORE 1024
MOVE pc acc
STORE 1025
JUMP CALL
LOAD 1030
STORE 11               # M(11) = 4111 (Pointer B)
PRINT 11

# Block C (n=5) als "Stopper" am Ende
LOADI 1
STORE 1026
LOADI 5
STORE 1030
LOADI *CALLOC
STORE 1024
MOVE pc acc
STORE 1025
JUMP CALL
LOAD 1030
STORE 12               # M(12) = 4124 (Pointer C)
PRINT 12

# --- PHASE 2: Internes Shrinking (mit Split) ---
# Wir verkleinern Block A auf n=2. Da Block B dahinter liegt,
# muss ein Split-Block entstehen.
LOADI 2
STORE 1026
LOAD 10
STORE 1030
LOADI 2
STORE 1031
LOADI *REALLOC
STORE 1024
MOVE pc acc
STORE 1025
JUMP CALL

PRINT 4096             # Header A: Sollte 5 sein (2+3)
PRINT 4101             # Header Split-Block: Sollte 8 sein (13-5)
PRINT 4102             # Status Split-Block: Sollte 0 (unused) sein

# --- PHASE 3: Shrinking am Ende (ohne Split) ---
# Wir verkleinern den Stopper-Block C. 
# Hier darf KEIN Split-Block entstehen.
LOADI 2
STORE 1026
LOAD 12
STORE 1030
LOADI 1
STORE 1031
LOADI *REALLOC
STORE 1024
MOVE pc acc
STORE 1025
JUMP CALL

PRINT 4122             # Header C: Sollte 4 sein (1+3)
# Prüfe, ob danach kein Header existiert (Heap-Ende)
PRINT 4095             # Top-Pointer: Muss immer noch 4122 sein

# --- PHASE 4: In-Place Expansion (Fusion) ---
# Wir vergrößern Block A wieder auf n=10.
# Er sollte den freien Split-Block von vorhin schlucken.
LOADI 2
STORE 1026
LOAD 10
STORE 1030
LOADI 10
STORE 1031
LOADI *REALLOC
STORE 1024
MOVE pc acc
STORE 1025
JUMP CALL

PRINT 1030             # Pointer A: Muss immer noch 4098 sein
PRINT 4096             # Header A: Sollte wieder 13 sein

# --- PHASE 5: Relocation (Umzug) ---
# Wir vergrößern A auf n=30. B versperrt den Weg.
# A muss hinter C umziehen.
LOADI 2
STORE 1026
LOAD 10
STORE 1030
LOADI 30
STORE 1031
LOADI *REALLOC
STORE 1024
MOVE pc acc
STORE 1025
JUMP CALL

LOAD 1030
STORE 13               # Neuen Pointer A in M(13) merken
PRINT 13               # Sollte > 4126 sein
# Check ob alter Platz frei ist
PRINT 4097             # Status bei 4097: Sollte 0 (free) sein

JUMP END

###### MIN METHOD ######
MIN:
    LOADIN 1 0              # IN1 is Frame-Pointer because Call Method puts it there
    STORE 1                 # x is now at M(1)
    LOADIN 1 1
    SUB 1                   # y - x
    JUMPC >= RETURN    # if (y-x)>=0: y>=x -> x is already at ReturnValue Adress
    LOADIN 1 1
    STOREIN 1 0             # Store y in ReturnValue Adress
    JUMP RETURN

###### MAX METHOD ######
MAX:
    LOADIN 1 0
    STORE 1                 # x is now at M(1)
    LOADIN 1 1
    SUB 1                   # y - x
    JUMPC <= RETURN    # if (y-x)<=0: y<=x -> x is already at ReturnValue Adress
    LOADIN 1 1
    STOREIN 1 0             # Store y in ReturnValue Adress
    JUMP RETURN

###### MUL METHOD ######
MUL:
    LOADIN 1 1
    STORE 2             # y Value
    JUMPC >= MULCHECKX
    LOADI 0
    SUB 2
    STORE 2             # If y<0 negate y
MULCHECKX:
    LOADIN 1 0
    STORE 1
    LOADIN 1 1
    JUMPC = RETURN      # if x is zero, zero is in return_adr, so return
    JUMPC >= MULSTART
    LOADI 0
    SUB 1
    STORE 1
MULSTART:
    LOADI 0             # Start with Zero
    STORE 3
MUL_LOOP:
    LOAD 2
    JUMPC <= MUL_DONE
    SUBI 1
    STORE 2
    LOAD 3
    ADD 1
    STORE 3
    JUMP MUL_LOOP
MUL_DONE:
    LOAD 0
    MOVE acc in1
    LOAD 3
    STOREIN 1 0
    JUMP RETURN
