SCHEDULER_SETUP:

    # Prepare
    LOADI 1
    STORE 1000             # Set PID to KERNEL_ENTRY
    STORE 1002              # Set Amount of Processes to 1

    # Create Boot-Stack
    LOADI 256               # Currently this supports a call-depth of ~20
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *MALLOC
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030               # <-- Boot-Stack Pointer
    STORE 1004
    MOVE ACC IN2

    # Create Process-Controll-Block for Kernel
    LOADI PCB_FRAME_SIZE
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *MALLOC
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    # Fill PCB with Kernel-Data
    LOAD 1030
    STORE 1001                  # Current PCB is this new one
    STORE 1003                  # Kernel PCV is this new one
    MOVE ACC IN1
    STOREIN 1 PCB_NEXT          # NEXT_PCB is this Block itself
    STOREIN 1 PCB_PREV          # PREV_PCB is also this Block itself
    LOADI 1
    STORE 1002                  # Amount of Processes, currently only Kernel
    STOREIN 1 PCB_PID           # Store PID on PCB
    LOADI 0
    STOREIN 1 PCB_WAIT_REASON   # No Wait reason
    STOREIN 1 PCB_WAIT_DATA     # No Wait Data
    STOREIN 1 PCB_STATE         # Store "Ready"-State on PCB
    STOREIN 1 PCB_EXIT_CODE
    STOREIN 1 PCB_MAILBOX
    STOREIN 1 PCB_MAILSENDER
    STOREIN 1 PCB_PARENT_PID    # Kernel has Parent PID 0!
    MOVE IN2 ACC
    STOREIN 1 PCB_STACK_POINTER     # Current Stack pointer (refresh on Process switch!)
    LOADI 2048
    STOREIN 1 PCB_STACK_BASE        # Stack Base
    LOADI 2047                      # Kernel-Stack 2048 - 4094
    STOREIN 1 PCB_STACKSIZE

    LOAD 1003                   # Kernel PCB
    STORE 1030
    LOADI *KERNEL_ENTRY         # Relative Offset to Kernel-Entry
    STORE 1
    MOVE PC ACC                 # Current Adress
    STORE 2
    LOAD 1
    ADD 2
    SUBI 2
    STORE 1031
    LOADI 0
    STORE 1032
    LOADI 3
    STORE 1026
    LOADI *SCHEDULER_PREPARE_PROCESS_STACK
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    # Switch to Kernel Stack
    LOAD 1001
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_POINTER
    MOVE ACC IN2

    # Now call Free-Method on Kernel Stack
    LOAD 1004
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *FREE
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOADI 0
    STORE 1004

    # Call Return and hope for the best
    JUMP RETURN                 # Due to manipulated Kernel stack, this should take execution to KERNEL_ENTRY


SCHEDULER_PREPARE_PROCESS_STACK:
    LOADIN 1 0                  # Pointer to PCB
    STORE 1
    LOADIN 1 1                  # Absolute Adress of Process Entry point
    STORE 2
    LOADIN 1 2                  # Amount of Process-Arguments
    STORE 3

    # Process-Stack is empty. Load Stack-Base and imitate Call-Frame
    LOAD 1
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_BASE
    MOVE ACC IN1                # Pointer to Stack

    # Fill IN1 and Working Registers with Zero
    LOADI 0
    STOREIN 1 0                 # IN1-Copy
    STOREIN 1 1                 # M(0)-Copy
    STOREIN 1 2
    STOREIN 1 3
    STOREIN 1 4
    STOREIN 1 5
    STOREIN 1 6
    STOREIN 1 7
    STOREIN 1 8
    STOREIN 1 9
    STOREIN 1 10                # M(9)-Copy
    MOVE IN1 ACC
    ADDI 11
    STORE 4                     # Pointer to new Stack

    # Copy Process-Arguments
    LOAD 0
    ADDI 3
    STORE 5                     # Pointer to arguments on own Stack

    LOAD 3
    STORE 7                     # Copy of Arg-Count

SCHEDULER_PREPARE_PROCESS_STACK_COLO:
    LOAD 7                      # Amount of Args
    JUMPC <= SCHEDULER_PREPARE_PROCESS_STACK_CODO
    SUBI 1
    STORE 7
    LOAD 5
    MOVE ACC IN1
    LOADIN 1 0
    STORE 6                     # Argument is on M(6)
    MOVE IN1 ACC
    ADDI 1
    STORE 5

    LOAD 4                      # Pointer to stack
    MOVE ACC IN1
    LOAD 6
    STOREIN 1 0
    MOVE IN1 ACC
    ADDI 1
    STORE 4
    JUMP SCHEDULER_PREPARE_PROCESS_STACK_COLO

SCHEDULER_PREPARE_PROCESS_STACK_CODO:
    # Copy of Arguments is done.
    LOAD 4
    MOVE ACC IN1                # Points to new Stack
    LOAD 3
    STOREIN 1 0                 # Amount of Arguments
    LOAD 2
    STOREIN 1 1                 # Absolute Adress of Process Entry Point
    MOVE IN1 ACC
    ADDI 2
    STORE 4                     # New Stack Pointer for PCB!

    LOAD 1
    MOVE ACC IN1
    LOAD 4
    STOREIN 1 PCB_STACK_POINTER
    JUMP RETURN



YIELD:
    LOAD 1001
    STORE 2                     # M(2) holds old PCB
    MOVE ACC IN1
    MOVE IN2 ACC
    STOREIN 1 PCB_STACK_POINTER

    # Check Stack for Overflow
    STORE 10
    LOADIN 1 PCB_STACK_BASE
    STORE 11
    LOAD 10
    SUB 11
    STORE 10                    # Holds Amount of filled Stack Cells
    LOADIN 1 PCB_STACKSIZE
    SUB 10
    JUMPC < EXCEPTION           # Call Exception because of Stack Overflow

    LOADIN 1 PCB_STATE
    SUBI PCB_STATE_RUNNING
    JUMPC != 3
    LOADI PCB_STATE_READY
    STOREIN 1 PCB_STATE

    # Check if Procces terminated itself
    LOADIN 1 PCB_STATE
    SUBI PCB_STATE_TERMINATED
    JUMPC != YIELD_CON_AFTER_TERMINATE

    # Check if Kernel is the process to Terminate
    LOAD 2
    SUB 1003
    JUMPC = 0                   # <-- This is shutdown!

    # Terminate old Process, need to switch to Kernel Stack before
    LOAD 1003
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_POINTER
    MOVE ACC IN2

    # Now call SCHEDULER_CLEAR_PROCESS
    LOAD 2
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *SCHEDULER_DELETE_PROCESS
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    # Done, now continue Search starting at Kernel, because old PCB might already be deleted
    LOAD 1003
    STORE 2

YIELD_CON_AFTER_TERMINATE:
    # Now goto next Process
    LOAD 2
    MOVE ACC IN1
    LOADIN 1 PCB_NEXT
    STORE 1                     # Pointer to PCB
YIELD_LOOP:
    LOAD 1
    MOVE ACC IN1
    LOADIN 1 PCB_STATE
    
    # Check if State is Waiting
    SUBI PCB_STATE_WAITING
    JUMPC != YIELD_NOWAIT
    # Check if reason is Input
    LOADIN 1 PCB_WAIT_REASON
    SUBI PCB_WAITING_INPUT
    JUMPC != YIELD_NOWAIT
    # Process is waiting for input. Check if Input exists
    MOVE IN1 ACC
    STORE 3
    LOAD 1003
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_BASE
    MOVE ACC IN1
    LOADIN 1 KERNEL_INBUF_COUNT
    STORE 4
    LOAD 3
    MOVE ACC IN1                # Reset Pointer to PCB
    LOAD 4
    JUMPC = YIELD_NOWAIT
    # Input available. Wake up process
    LOADI PCB_STATE_READY
    STOREIN 1 PCB_STATE
    LOADI 0
    STOREIN 1 PCB_WAIT_REASON

YIELD_NOWAIT:
    LOADIN 1 PCB_STATE
    SUBI PCB_STATE_READY
    JUMPC != YIELD_LOOK_NEXT
    LOAD 1
    STORE 1001                  # Update PCB Pointer
    LOADIN 1 PCB_PID
    STORE 1000                  # Update PID
    LOADI PCB_STATE_RUNNING
    STOREIN 1 PCB_STATE
    LOADIN 1 PCB_STACK_POINTER
    MOVE ACC IN2

    JUMP RETURN

YIELD_LOOK_NEXT:
    LOADIN 1 PCB_NEXT
    STORE 1
    JUMP YIELD_LOOP

##### WAIT #####
WAIT:
    LOADIN 1 0
    STORE 1                     # PID of Child

    # Catch Wait for itself
    LOAD 1
    SUB 1000
    JUMPC = WAIT_RETURN_ERROR
    
    # Get PCB
    LOAD 1
    STORE 1030
    LOADI PCB_PID
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *SCHEDULER_GET_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    STORE 2

    JUMPC = WAIT_RETURN_ERROR
    LOAD 1000
    SUBI 1
    JUMPC = WAIT_ALLOWED
    LOAD 2
    MOVE ACC IN1
    LOADIN 1 PCB_PARENT_PID
    SUB 1000
    JUMPC != WAIT_RETURN_ERROR
WAIT_ALLOWED:
    LOAD 2
    MOVE ACC IN1
    LOADIN 1 PCB_STATE
    SUBI PCB_STATE_TERMINATED
    JUMPC != WAIT_GO_SLEEP
    LOADIN 1 PCB_EXIT_CODE
    STORE 3                     # 3 holds Exit Code

    # Remove PCB
    LOAD 2
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *SCHEDULER_REMOVE_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOAD 0
    MOVE ACC IN1
    LOAD 3
    STOREIN 1 0
    LOADI 0
    STOREIN 1 1
    JUMP RETURN

WAIT_GO_SLEEP:
    # Set State to Waiting for Child
    LOAD 1001
    MOVE ACC IN1
    LOADI PCB_STATE_WAITING
    STOREIN 1 PCB_STATE
    LOADI PCB_WAITING_CHILD
    STOREIN 1 PCB_WAIT_REASON
    LOAD 1
    STOREIN 1 PCB_WAIT_DATA

    # Yield
    LOADI 0
    STORE 1026
    LOADI *YIELD
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOADIN 1 PCB_WAIT_DATA
    STORE 3
    LOAD 0
    MOVE ACC IN1
    LOAD 3
    STOREIN 1 0
    LOADI 0
    STOREIN 1 1
    JUMP RETURN


WAIT_RETURN_ERROR:
    LOAD 0
    MOVE ACC IN1
    LOADI 1
    STOREIN 1 1                 # Mark that Child does not exist or Child is itself
    JUMP RETURN

##### SEND #####
SEND:
    LOADIN 1 0
    STORE 1                     # PID of Receiver
    LOADIN 1 1
    STORE 2                     # Message
    LOADIN 1 2
    STORE 4                     # If 0, dont go to sleep
    LOADI SEND_OKAY
    STORE 5                     # Error Flag

    # Catch Send to itself
    LOAD 1
    SUB 1000
    JUMPC != SEND_CON
    LOADI SEND_TOSELF
    STOREIN 1 0
    JUMP RETURN

SEND_CON:
    # Get PCB
    LOAD 1
    STORE 1030
    LOADI PCB_PID
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *SCHEDULER_GET_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    STORE 3

    # Check if Mailbox is empty
    JUMPC != 4
    LOADI SEND_NO_REC
    STORE 5
    JUMP SEND_RETURN
    MOVE ACC IN1
    LOADIN 1 PCB_STATE
    SUBI PCB_STATE_TERMINATED
    JUMPC != 4
    LOADI SEND_REC_TERM
    STORE 5
    JUMP SEND_RETURN
    LOADIN 1 PCB_MAILSENDER
    JUMPC != SEND_MAILFULL

    # Store Mail
    LOAD 1000                   # Own PID
    STOREIN 1 PCB_MAILSENDER
    LOAD 2
    STOREIN 1 PCB_MAILBOX

    # Check if Receiver is waiting for Mail, wake up if so
    LOADIN 1 PCB_STATE
    SUBI PCB_STATE_WAITING
    JUMPC != SEND_SUCCES
    LOADIN 1 PCB_WAIT_REASON
    SUBI PCB_WAITING_MAIL
    JUMPC != SEND_SUCCES
    
    LOADI PCB_STATE_READY
    STOREIN 1 PCB_STATE
    LOADI 0
    STOREIN 1 PCB_WAIT_REASON
    JUMP SEND_SUCCES

SEND_MAILFULL:
    LOAD 4
    JUMPC != 4
    LOADI SEND_REC_FULL
    STORE 5
    JUMP SEND_RETURN
    LOAD 1001
    MOVE ACC IN1
    LOADI PCB_STATE_WAITING
    STOREIN 1 PCB_STATE
    LOADI PCB_WAITING_OUTPUT
    STOREIN 1 PCB_WAIT_REASON
    LOAD 1
    STOREIN 1 PCB_WAIT_DATA

    # Yield
    LOADI 0
    STORE 1026
    LOADI *YIELD
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOADIN 1 PCB_WAIT_DATA
    JUMPC != 4
    LOADI SEND_REC_TERM
    STORE 5
    JUMP SEND_RETURN
    LOAD 0
    MOVE ACC IN1
    JUMP SEND

SEND_SUCCES:
    LOADI SEND_OKAY
    STORE 5
SEND_RETURN:
    LOAD 0
    MOVE ACC IN1
    LOAD 5
    STOREIN 1 0
    JUMP RETURN

    LOAD 0

    LOAD 2

    STOREIN 1 PCB_STATE
    LOADI PCB_WAITING_MAIL
    STOREIN 1 PCB_WAIT_REASON

    # Yield
    LOADI 0

    LOADI *YIELD
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    MOVE ACC IN1

    MOVE ACC IN1

    STOREIN 1 0
    STOREIN 1 1
    JUMP RETURN


SCHEDULER_DELETE_PROCESS:
    LOAD 1000
    STORE 1                     # Store current PID
    LOADI 1
    STORE 1000                  # Switch to Kernel-Mode
    LOADIN 1 0
    STORE 2                     # PCP to delete

    # Clear Stack and Heap
    LOAD 2
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *SCHEDULER_CLEAR_PROCESS
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOAD 2
    MOVE ACC IN1
    LOADIN 1 PCB_PID
    STORE 3

SCHEDULER_DELETE_PROCESS_CL:
    # Look up every Child and force-terminate it
    LOAD 3
    STORE 1030
    LOADI PCB_PARENT_PID
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *SCHEDULER_GET_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    JUMPC = SCHEDULER_DELETE_PROCESS_CD
    
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *SCHEDULER_DELETE_PROCESS_FORCE
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    JUMP SCHEDULER_DELETE_PROCESS_CL

SCHEDULER_DELETE_PROCESS_CD:
    # Wake Output-Waiting Processes
    LOAD 3
    STORE 1030
    LOADI 1
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *SCHEDULER_WAKE_SENDERS
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    # Check if Parent is Waiting
    LOAD 2
    MOVE ACC IN1
    LOADIN 1 PCB_PARENT_PID
    STORE 1030
    # If Parent is Kernel, directly delete PCB
    SUBI 1
    JUMPC = SCHEDULER_DELETE_PROCESS_PK
    LOADI PCB_PID
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *SCHEDULER_GET_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOAD 1030
    STORE 4                     # 4 holds PCB of Parent
    MOVE ACC IN1
    LOADIN 1 PCB_STATE
    SUBI PCB_STATE_WAITING
    JUMPC != SCHEDULER_DELETE_PROCESS_SP
    LOADIN 1 PCB_WAIT_REASON
    SUBI PCB_WAITING_CHILD
    JUMPC != SCHEDULER_DELETE_PROCESS_SP
    LOADIN 1 PCB_WAIT_DATA
    SUB 3
    JUMPC != SCHEDULER_DELETE_PROCESS_SP

    # Parent is waiting on this Process!
    # Wakeup Parent
    LOAD 2
    MOVE ACC IN1
    LOADIN 1 PCB_EXIT_CODE
    STORE 5
    LOAD 4
    MOVE ACC IN1
    LOAD 5
    STOREIN 1 PCB_WAIT_DATA
    LOADI 0
    STOREIN 1 PCB_WAIT_REASON
    LOADI PCB_STATE_READY
    STOREIN 1 PCB_STATE

SCHEDULER_DELETE_PROCESS_PK:
    # Remove PCB
    LOAD 2
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *SCHEDULER_REMOVE_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    JUMP SCHEDULER_DELETE_PROCESS_D

SCHEDULER_DELETE_PROCESS_SP:

SCHEDULER_DELETE_PROCESS_D:
    LOAD 1
    STORE 1000
    JUMP RETURN

##### SCHEDULER_DELETE_PROCESS_FORCE #####
SCHEDULER_DELETE_PROCESS_FORCE:
# Force is used to imideatly kill and delete a Process and its Childrends. No Handling of Exit-Code
    LOADIN 1 0
    STORE 1             # Root PCB to delete
    STORE 2             # v
SCHEDULER_DELETE_PROCESS_FORCE_L:
    # Get any Children Process
    LOAD 2
    MOVE ACC IN1
    LOADIN 1 PCB_PID
    STORE 1030
    LOADI PCB_PARENT_PID
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *SCHEDULER_GET_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030           # PCB that has this PID as its Parent
    STORE 3             # c
    JUMPC = SCHEDULER_DELETE_PROCESS_FORCE_NN

    LOAD 3
    STORE 2
    JUMP SCHEDULER_DELETE_PROCESS_FORCE_L

SCHEDULER_DELETE_PROCESS_FORCE_NN:
    LOAD 2
    MOVE ACC IN1
    LOADIN 1 PCB_PARENT_PID
    STORE 1030
    LOADI PCB_PID
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *SCHEDULER_GET_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    STORE 4

    # Wake up every sender to this process and cause Error
    LOAD 2
    MOVE ACC IN1
    LOADIN 1 PCB_PID
    STORE 1030
    LOADI 1
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *SCHEDULER_WAKE_SENDERS
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    # Clear Process
    LOAD 2
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *SCHEDULER_CLEAR_PROCESS
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    # Remove PCB
    LOAD 2
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *SCHEDULER_REMOVE_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    # Check if this PCB was Root
    LOAD 2
    SUB 1
    JUMPC = RETURN 

    # Go Back to Parent
    LOAD 4
    STORE 2
    JUMP SCHEDULER_DELETE_PROCESS_FORCE_L


SCHEDULER_CLEAR_PROCESS:
# Frees the Stack and Heap of a PCB. Only Call in Kernel Mode!
    LOADIN 1 0
    STORE 1             # PCB

    # Free terminated Stack
    LOAD 1
    JUMPC = SCHEDULER_CLEAR_PROCESS_SZ
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_BASE
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *FREE
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

SCHEDULER_CLEAR_PROCESS_SZ:
    LOADIN 1 PCB_PID
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *FREE_PID
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOADI 0
    STOREIN 1 PCB_STACK_BASE
    STOREIN 1 PCB_STACK_POINTER
    STOREIN 1 PCB_STACKSIZE
    JUMP RETURN

SCHEDULER_REMOVE_PCB:
    LOADIN 1 0
    STORE 1

    # Remove PCB out of Linked List
    MOVE ACC IN1
    LOADIN 1 PCB_PID
    STORE 7

    LOADIN 1 PCB_NEXT
    STORE 4
    LOADIN 1 PCB_PREV
    STORE 5
    MOVE ACC IN1
    LOAD 4
    STOREIN 1 PCB_NEXT
    MOVE ACC IN1
    LOAD 5
    STOREIN 1 PCB_PREV

    # Free terminated PCB
    LOAD 1
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *FREE
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    JUMP RETURN

##### SCHEDULER_GET_PCB: #####
SCHEDULER_GET_PCB:
    LOADIN 1 0
    STORE 1             # Value to look for
    LOADIN 1 1
    STORE 2             # Index where to find the Value
    LOAD 1003
    MOVE ACC IN1

SCHEDULER_GET_PCB_L:
    MOVE IN1 ACC
    STORE 3
    ADD 2
    MOVE ACC IN1
    LOADIN 1 0
    SUB 1
    JUMPC = SCHEDULER_GET_PCB_D
    LOAD 3
    MOVE ACC IN1
    LOADIN 1 PCB_NEXT
    MOVE ACC IN1
    SUB 1003
    JUMPC = SCHEDULER_GET_PCB_N
    JUMP SCHEDULER_GET_PCB_L

SCHEDULER_GET_PCB_N:
    LOADI 0
    STORE 3

SCHEDULER_GET_PCB_D:
    LOAD 0
    MOVE ACC IN1
    LOAD 3
    STOREIN 1 0
    JUMP RETURN

##### SCHEDULER_WAKE_SENDERS #####
SCHEDULER_WAKE_SENDERS:
    LOADIN 1 0
    STORE 1                 # PID of Receiver. 
    LOADIN 1 1
    STORE 2                 # If the Receiver is/was Terminated, set this Flag to 1
    LOAD 1003
    MOVE ACC IN1
SCHEDULER_WAKE_SENDERS_L:
    LOADIN 1 PCB_NEXT
    MOVE ACC IN1            # Goto next PCB (firs pcb is kernel, cant be waiting)
    SUB 1003
    JUMPC = RETURN
    LOADIN 1 PCB_STATE
    SUBI PCB_STATE_WAITING
    JUMPC != SCHEDULER_WAKE_SENDERS_L
    LOADIN 1 PCB_WAIT_REASON
    SUBI PCB_WAITING_OUTPUT
    JUMPC != SCHEDULER_WAKE_SENDERS_L
    LOADIN 1 PCB_WAIT_DATA
    SUB 1
    JUMPC != SCHEDULER_WAKE_SENDERS_L

    # This Process needs to be woken up!
    LOADI PCB_STATE_READY
    STOREIN 1 PCB_STATE
    LOADI 0
    STOREIN 1 PCB_WAIT_REASON
    LOAD 2
    JUMPC = SCHEDULER_WAKE_SENDERS_L
    LOADI 0
    STOREIN 1 PCB_WAIT_DATA
    JUMP SCHEDULER_WAKE_SENDERS_L


SCHEDULER_FIND_FREE_PID:
    LOAD 1003
    MOVE ACC IN1            # Kernel-PCB
    LOADI 2
    STORE 1                 # expected PID 
SCHEDULER_FIND_FREE_PID_L:
    MOVE IN1 ACC
    STORE 2                 
    LOADIN 1 PCB_NEXT
    MOVE ACC IN1            # next PCB
    SUB 1003
    JUMPC = SCHEDULER_FIND_FREE_PID_D   # If PCB is Kernel again, stop
    LOADIN 1 PCB_PID
    SUB 1
    JUMPC != SCHEDULER_FIND_FREE_PID_D  # PID is bigger than its index, so index is free
    LOAD 1
    ADDI 1
    STORE 1
    JUMP SCHEDULER_FIND_FREE_PID_L

SCHEDULER_FIND_FREE_PID_D:
    LOAD 0
    MOVE ACC IN1
    LOAD 1
    STOREIN 1 0              # new PID
    JUMP RETURN
