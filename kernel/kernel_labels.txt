KERNEL_ENTRY:
    MOVE IN2 ACC
    STORE 1030                  # Start of the global cells
    ADDI KERNEL_STACKSPACE
    MOVE ACC IN2                # Allocate first KERNEL_STACKSPACE Cells of Stack for process-global values
                                # (before MEMSET, otherwise its Call-Frame lies inside the cleared area)
    LOADI KERNEL_STACKSPACE
    STORE 1031
    LOADI 0
    STORE 1032
    LOADI 3
    STORE 1026
    LOADI *MEMSET
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    

    # get Input Buffer
    LOADI KERNEL_INBUF_CONSTSIZE
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *CALLOC
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1001
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_BASE
    MOVE ACC IN1
    LOAD 1030
    STOREIN 1 KERNEL_INBUF_BASE
    STOREIN 1 KERNEL_INBUF_INCELL
    STOREIN 1 KERNEL_INBUF_OUTCELL
    LOADI 0
    STOREIN 1 KERNEL_INBUF_COUNT

    # Now create ECHO Process for Testing
    LOADI 1                 
    STORE 1030              # Store Parent PID

    LOADI *ECHO_ENTRY        # Relative Offset to Kernel-Entry
    STORE 1
    MOVE PC ACC                 # Current Adress
    ADD 1
    SUBI 2
    STORE 1031              # Entry Point adress

    LOADI 0
    STORE 1032              # Amount of Process Args
    LOADI 3
    STORE 1026
    LOADI *CREATE_TASK
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

##### KERNEL_READ_INPUT #####
KERNEL_READ_INPUT:
    # Check if Input Buffer is already full
    LOAD 1003
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_BASE
    MOVE ACC IN1
    LOADIN 1 KERNEL_INBUF_COUNT
    SUBI KERNEL_INBUF_CONSTSIZE
    JUMPC >= KERNEL_READ_INPUT_FULL
    
    # Buffer isn't full. Try to read next input
    READ KERNEL_READ_INPUT_DONE          # Jumps to Flag if there is no Input, otherwise contiues
    STORE 0                              # Key (ACC) would be lost by the LOADINs below
    LOADIN 1 KERNEL_INBUF_BASE
    STORE 1
    LOADIN 1 KERNEL_INBUF_INCELL
    STORE 2
    MOVE ACC IN1
    LOAD 0
    STOREIN 1 0

    # Update Ringbuffer-Pointer
    LOAD 2                                  
    ADDI 1
    STORE 2                                 # Absolute Adresse
    SUB 1                                   # Now is relative Index
    SUBI KERNEL_INBUF_CONSTSIZE
    JUMPC < KERNEL_READ_INPUT_NOWRAP
    LOAD 1
    STORE 2
KERNEL_READ_INPUT_NOWRAP:
    LOAD 1003
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_BASE
    MOVE ACC IN1
    LOAD 2
    STOREIN 1 KERNEL_INBUF_INCELL
    LOADIN 1 KERNEL_INBUF_COUNT
    ADDI 1
    STOREIN 1 KERNEL_INBUF_COUNT
KERNEL_READ_INPUT_DONE:
    JUMP RETURN

KERNEL_READ_INPUT_FULL:
    # Design choice to make. Until now, dont read input buffer
    JUMP RETURN

##### READ_KEY #####
READ_KEY:
    # This Function can be called by any Process, so Kernel-Context is not automatically provided!
    # Check if there is any Key in the buffer. Otherwise, do a YIELD
    LOAD 1003
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_BASE
    MOVE ACC IN1
    LOADIN 1 KERNEL_INBUF_COUNT
    STORE 3
    JUMPC <= READ_KEY_EMPTY

    LOADIN 1 KERNEL_INBUF_BASE
    STORE 1
    LOADIN 1 KERNEL_INBUF_OUTCELL
    STORE 2
    MOVE ACC IN1
    LOADIN 1 0
    STORE 5                         # Key Value is on M(5) now!

    LOAD 2
    ADDI 1
    STORE 2
    SUB 1
    SUBI KERNEL_INBUF_CONSTSIZE
    JUMPC < READ_KEY_NOWRAP
    LOAD 1
    STORE 2
READ_KEY_NOWRAP:
    LOAD 1003
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_BASE
    MOVE ACC IN1
    LOAD 2
    STOREIN 1 KERNEL_INBUF_OUTCELL
    LOADIN 1 KERNEL_INBUF_COUNT
    SUBI 1
    STOREIN 1 KERNEL_INBUF_COUNT
    LOAD 0
    MOVE ACC IN1
    LOAD 5
    STOREIN 1 0
    JUMP RETURN

READ_KEY_EMPTY:
    # If empty, do a Yield after Setting State to Waiting for Input
    LOAD 1001
    MOVE ACC IN1
    LOADI PCB_STATE_WAITING
    STOREIN 1 PCB_STATE
    LOADI PCB_WAITING_INPUT
    STOREIN 1 PCB_WAIT_REASON
    LOADI 0
    STORE 1026 
    LOADI *YIELD
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    JUMP READ_KEY

##### KERNEL_INSERT_PCB #####
KERNEL_INSERT_PCB:
    LOADIN 1 0
    STORE 1                     # Pointer to new PCB
    MOVE ACC IN1
    LOADIN 1 PCB_PID
    STORE 2                     # PID of new PCB
    LOAD 1003
    MOVE ACC IN1                # Start at Kernel-PCB
KERNEL_INSERT_PCB_L:
    MOVE IN1 ACC
    STORE 3                     # current = candidate for predecessor
    LOADIN 1 PCB_NEXT
    MOVE ACC IN1                # IN1 = next
    SUB 1003
    JUMPC = KERNEL_INSERT_PCB_D # next is Kernel again: insert at the end
    LOADIN 1 PCB_PID
    SUB 2
    JUMPC > KERNEL_INSERT_PCB_D # next has bigger PID: insert before it
    JUMP KERNEL_INSERT_PCB_L

KERNEL_INSERT_PCB_D:
    MOVE IN1 ACC
    STORE 4                     # old next
    LOAD 1
    STOREIN 1 PCB_PREV          # old next.prev = new
    LOAD 3
    MOVE ACC IN1
    LOAD 1
    STOREIN 1 PCB_NEXT          # predecessor.next = new
    MOVE ACC IN1                # IN1 = new
    LOAD 3
    STOREIN 1 PCB_PREV          # new.prev = predecessor
    LOAD 4
    STOREIN 1 PCB_NEXT          # new.next = old next
    LOAD 1002
    ADDI 1
    STORE 1002
    JUMP RETURN

KERNEL_CREATE_PCB:
    LOADIN 1 0
    STORE 1             # PID

    # Allocate Heap for PCB
    LOADI PCB_FRAME_SIZE
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *MALLOC
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    STORE 2
    MOVE ACC IN1
    
    # Prepare PCB
    LOAD 1
    STOREIN 1 PCB_PID
    LOADI 0
    STOREIN 1 PCB_STATE
    STOREIN 1 PCB_WAIT_REASON
    STOREIN 1 PCB_WAIT_DATA
    STOREIN 1 PCB_EXIT_CODE
    STOREIN 1 PCB_MAILBOX
    STOREIN 1 PCB_MAILSENDER

    # Allocate Heap for Stack
    LOADI DEFAULT_PROCESS_STACKSIZE
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *CALLOC
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    STOREIN 1 PCB_STACK_BASE
    STOREIN 1 PCB_STACK_POINTER
    LOADI DEFAULT_PROCESS_STACKSIZE
    STOREIN 1 PCB_STACKSIZE

    LOAD 0
    MOVE ACC IN1
    LOAD 2
    STOREIN 1 0
    JUMP RETURN

##### CREATE_TASK #####
CREATE_TASK:
    # Can be called out of any Context!
    LOAD 1000
    STORE 1             # Save current PID
    LOADI 1
    STORE 1000          # Switch to Kernel Mode

    LOADIN 1 0
    STORE 2             # Parent PID
    LOADIN 1 1
    STORE 3             # Absolute Adress of Entry Point
    LOADIN 1 2
    STORE 4             # Amount of Process Arguments

    # Check if Parent-PID is Valid (either PID of current Process or Kernel)
    LOAD 2
    SUB 1
    JUMPC = CREATE_TASK_VALPID
    LOAD 2
    SUBI 1
    JUMPC = CREATE_TASK_VALPID
    LOAD 1
    STORE 1000
    JUMP EXCEPTION

CREATE_TASK_VALPID:
    LOADI 1
    STORE 1026
    LOADI 0
    STORE 1030
    LOADI *SCHEDULER_FIND_FREE_PID
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    STORE 9             # PID of new Process

    # Create & Prepare PCB
    LOAD 9
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *KERNEL_CREATE_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOAD 1030       
    STORE 5                 # Pointer to new PCB

    # Set Parent and Exit-Code
    MOVE ACC IN1
    LOAD 2                  # Parent PID
    STOREIN 1 PCB_PARENT_PID
    LOAD 9
    STORE 2

    # Now prepare new Stack.
    LOAD 5
    STORE 1030              # PCB
    LOAD 3
    STORE 1031              # Absolute adress of entry Point
    LOAD 4
    STORE 1032              # Amount of Process-Arguments
    STORE 6                 # Working Copy

    LOAD 0
    ADDI 3
    STORE 7                 # Pointer to first Argument source
    LOADI 1033
    STORE 8                 # Pointer to first Argument 
    
CREATE_TASK_CAL:
    LOAD 6
    JUMPC <= CREATE_TASK_CAD
    SUBI 1
    STORE 6
    LOAD 7
    MOVE ACC IN1
    ADDI 1
    STORE 7
    LOADIN 1 0
    STORE 9
    LOAD 8
    MOVE ACC IN1
    ADDI 1
    STORE 8
    LOAD 9
    STOREIN 1 0
    JUMP CREATE_TASK_CAL

CREATE_TASK_CAD:
    LOADI 3
    ADD 4
    STORE 1026
    LOADI *SCHEDULER_PREPARE_PROCESS_STACK
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOAD 5
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *KERNEL_INSERT_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1
    STORE 1000

    LOAD 0
    MOVE ACC IN1
    LOAD 2              # New PID
    STOREIN 1 0
    JUMP RETURN

##### EXIT #####
EXIT:
    LOADIN 1 0
    STORE 1             # Exit-Code
    LOAD 1001           # Current PCB
    MOVE ACC IN1
    LOAD 1
    STOREIN 1 PCB_EXIT_CODE
    LOADI PCB_STATE_TERMINATED
    STOREIN 1 PCB_STATE

    # Call YIELD
    LOADI 0
    STORE 1026
    LOADI *YIELD
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

##### TERMINATE #####
TERMINATE:
    LOAD 1000
    STORE 1             # PID of Caller
    LOADIN 1 0
    STORE 2             # PID of Process to Terminate
    SUB 1
    JUMPC = RETURN      # Cannot Terminate itself
    LOAD 2
    SUBI 1
    JUMPC = RETURN      # Cannot Terminate Kernel (only Exit allowed -> Shutdown)

    # Get PCB of Process to Terminate
    LOAD 2
    STORE 1030
    LOADI PCB_PID
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *SCHEDULER_GET_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    JUMPC = RETURN      # If PCB is Zero, Process does not exist
    STORE 3
    MOVE ACC IN1

    LOADIN 1 PCB_STATE
    SUBI PCB_STATE_TERMINATED
    JUMPC = RETURN              # If Process is already Terminated, no need to terminate

    # Check if Parent is Kernel
    LOAD 1
    SUBI 1
    JUMPC = TERMINATE_ALLOWED   # Kernel can Terminate every Process

    LOADIN 1 PCB_PARENT_PID
    SUB 1
    JUMPC != RETURN
    
TERMINATE_ALLOWED:
    # Termination is allowed
    LOAD 3
    MOVE ACC IN1
    LOADI -1
    STOREIN 1 PCB_EXIT_CODE
    LOADI PCB_STATE_TERMINATED
    STOREIN 1 PCB_STATE

    LOAD 3
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *SCHEDULER_DELETE_PROCESS
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    JUMP RETURN

##### SHUTDOWN #####
SHUTDOWN:
    # Terminate every Child (so every other Task than this)
    # Than Exit this Task. YIELD will then Shutdown System.
    LOAD 1000
    SUBI 1
    JUMPC = SHUTDOWN_KERNEL
    
    # Switch to Kernel Stack because any other Stacks will be terminated
    LOADI 1
    STORE 1000
    LOAD 1003
    STORE 1001
    MOVE ACC IN1
    LOADIN 1 PCB_STACK_POINTER
    MOVE ACC IN2
SHUTDOWN_KERNEL:
    LOADI 2
    STORE 1026
    LOADI 1                 # Search for Kernel
    STORE 1030
    LOADI PCB_PARENT_PID    # as Parent PID
    STORE 1031
    LOADI *SCHEDULER_GET_PCB
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    JUMPC = SHUTDOWN_DONE
    STORE 1
    MOVE ACC IN1
    LOADIN 1 PCB_PID
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *TERMINATE
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    JUMP SHUTDOWN_KERNEL

SHUTDOWN_DONE:
    LOADI 0
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *EXIT
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
