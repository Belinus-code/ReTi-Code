###### MALLOC METHOD ######
MALLOC:
    LOADIN 1 0
    STORE 1                 # Size of to alloc memory
    LOADI 4096              # Points of first frame
    MOVE acc in1
MALLOC_SEARCH_LOOP:
    MOVE in1 acc
    SUB 4095
    JUMPC > MALLOC_HEAP_END
    LOADIN 1 1              # Frame used?
    JUMPC > MALLOC_GO_NEXT
    LOADIN 1 0
    SUB 1
    SUBI 3
    JUMPC >= MALLOC_FOUND
MALLOC_GO_NEXT:
    LOADIN 1 0              # Load Frame Size
    STORE 2                 # Store Frame Size to M(2)
    MOVE in1 acc
    ADD 2                   # Frame Pointer + Frame Size 
    MOVE acc in1            # Point to next Frame
    JUMP MALLOC_SEARCH_LOOP
MALLOC_FOUND:
    # If found but not at heap end, check if not last block before end
    # Check if block could be splitt in two, second block would need 4
    # Right now Data_len_old - Data_len_needed of current block is in acc
    SUBI 3                  # Overhead of split frame would need 3
    JUMPC <= MALLOC_ALLOC    # If no Jump then split block
    LOADIN 1 0
    STORE 3                 # Store original Frame Size in M(3)
    LOAD 1
    ADDI 3
    STOREIN 1 0             # Change Frame Size to new Size
    STORE 2                 # Save new Frame Size
    LOADI 1
    STOREIN 1 1             # Mark Frame as used

    # Now store new Frame size at end of Frame
    MOVE in1 acc
    STORE 4                 # Store Pointer of this Frame
    ADD 2                   # Points to next Frame
    MOVE acc in1
    LOAD 2
    STOREIN 1 -1            # Store new Frame size at end of Frame
    LOAD 4
    MOVE acc in1            # Restore Pointer to Frame Start

    # Before setting up splitt Block between this and next Block,
    # Check if this Block is last Block before Top of Heap
    MOVE in1 acc
    SUB 4095
    JUMPC >= MALLOC_FINISH   # Zero if this is last block. If positive, smth went wrong

    # Now create split Block after this one
    LOADIN 1 0              # New Frame Size
    STORE 2
    MOVE in1 acc
    ADD 2
    MOVE acc in1            # Points to new split Frame
    LOADI 0
    STOREIN 1 1             # Mark as unused
    # Calculate split Block size
    LOAD 3                  # Original Block Size
    SUBI 3
    SUB 1                   
    STOREIN 1 0             # Store split block size
    STORE 3
    
    # Store split Block size at end of split Frame
    MOVE in1 acc
    ADD 3                   # Points to next Frame
    MOVE acc in1
    LOAD 3
    STOREIN 1 -1            # Store new Frame size at end of Frame
    JUMP MALLOC_FINISH
    # Done with this branch

MALLOC_HEAP_END:
    LOAD 1                  # Args count
    ADDI 3
    STOREIN 1 0             # Store Frame Size on Heap
    STORE 2

    # Now store new Frame size at end of Frame
    MOVE in1 acc
    STORE 4                 # Store Pointer of this Frame
    STORE 4095              # Update Heap Top Pointer
    ADD 2                   # Points to next Frame
    MOVE acc in1
    LOAD 2
    STOREIN 1 -1            # Store new Frame size at end of Frame
    LOAD 4
    MOVE acc in1            # Restore Pointer to Frame Start

# Alloc without splitting
MALLOC_ALLOC:
    LOADI 1
    STOREIN 1 1             # Mark Frame as used
    MOVE in1 acc
    STORE 4
    JUMP MALLOC_FINISH

MALLOC_FINISH:
    LOAD 0
    MOVE acc in1                # Load Pointer to return arg
    LOAD 4                      # Pointer to Heap Frame
    ADDI 2                      # Now point to first arg in heap Frame
    STOREIN 1 0
    JUMP RETURN

###### FREE METHOD ######
FREE:
    LOADIN 1 0                  # Heap Pointer to data
    SUBI 2                      # Change pointer to frame start
    STORE 1
    MOVE acc in1                # Heap Frame Pointer in IN1
    LOADI 0
    STOREIN 1 1                 # Mark Frame as unused

    # Check if Frame is last Frame on heap
    LOAD 1
    SUB 4095
    JUMPC >= FREE_LAST_FRAME

    # Now check if Frame behind is unused
    LOADIN 1 0
    STORE 2
    MOVE in1 acc
    ADD 2
    MOVE acc in1
    LOADIN 1 1                  # Is next Frame used?
    JUMPC > FREE_CHECK_UNDER    # JUMP if used
    # Fuse this and next Frame
    LOADIN 1 0
    STORE 3                     # Store size of next Block
    LOAD 1
    MOVE acc in1                # Restore frame pointer to Block
    LOAD 3                      # Load size of next Block
    ADD 2                       # Add size of this Block
    STOREIN 1 0                 # Update size of Block
    STORE 2
    MOVE in1 acc
    ADD 2
    MOVE acc in1                # Jump to end of fused Frame
    LOAD 2
    STOREIN 1 -1                # Update Frame Length at end of Frame

FREE_CHECK_UNDER:
    MOVE in1 acc
    SUB 2
    MOVE acc in1                # Restore Pointer to Frame
    # Before checking Block before, ensure that this Block is not at M(4096)
    SUBI 4096
    JUMPC <= RETURN

    LOADIN 1 -1
    STORE 3
    MOVE in1 acc
    SUB 3
    MOVE acc in1
    LOADIN 1 1
    JUMPC > RETURN              # If Block before is used, Return
    LOAD 3
    ADD 2
    STOREIN 1 0
    STORE 2
    MOVE in1 acc
    ADD 2
    MOVE acc in1                # Jump to end of fused Frame
    LOAD 2
    STOREIN 1 -1                # Update Frame Length at end of Frame
    JUMP RETURN


FREE_LAST_FRAME:
    # Update Heap Top Pointer
    LOAD 4095                   # Heap Top Pointer
    SUBI 4096
    JUMPC <= FREE_BOTTOM_STACK

    # Now check if Frame before is unused. Delete both if so
    LOADIN 1 -1
    STORE 3
    MOVE in1 acc
    SUB 3
    MOVE acc in1
    LOADIN 1 1
    JUMPC = FREE_LAST_FRAME_CON
    MOVE in1 acc
    ADD 3
    MOVE acc in1

FREE_LAST_FRAME_CON:
    MOVE in1 acc
    SUBI 4096
    JUMPC <= FREE_BOTTOM_STACK
    LOADIN 1 -1
    STORE 3
    MOVE in1 acc
    SUB 3
    STORE 4095                  # Store updated Heap top Pointer
    JUMP RETURN

FREE_BOTTOM_STACK:
    LOADI 4095
    STORE 4095
    JUMP RETURN

###### REALLOC METHOD ######
REALLOC:
    LOADI 0
    STORE 4                     # Reset Flag in M(4)
    LOADIN 1 1                  # Wanted Size
    STORE 2                     
    LOADIN 1 0                  # Heap Pointer to data
    SUBI 2                      # Change pointer to frame start
    STORE 1
    MOVE acc in1                # Heap Frame Pointer in IN1
    LOADIN 1 0
    SUBI 3                      # 3 cells needed for Header/Footer
    SUB 2                       # Space needed for new Size
    JUMPC < REALLOC_NO_SHRINK   # If old-space - new_space >= 0, reuse block. 0 = perfect fit
    # Old Block Reusable. Check if Splitting makes sense and if frame is last on heap
    STORE 3
    MOVE in1 acc
    SUB 4095
    JUMPC < REALLOC_CON_SPLIT
    LOADI 1
    STORE 4                     # Store Flag to resize Block but not to create new Split Block

REALLOC_CON_SPLIT:
    LOAD 3
    SUBI 3                      # Check if Splitting Frame is possible
    JUMPC <= RETURN             # If split useless, return. Pointer already on arg stack

    # Split Frames. Use Code from Malloc
    LOADIN 1 0
    STORE 3                 # Store original Frame Size in M(3)
    LOAD 2                  # Load wanted size
    ADDI 3                  # Add overhead size
    STOREIN 1 0             # Change Frame Size to new Size
    STORE 2                 # Save new Frame Size

    # Now store new Frame size at end of Frame
    MOVE in1 acc
    ADD 2                   # Points to next Frame
    MOVE acc in1
    LOAD 2
    STOREIN 1 -1            # Store new Frame size at end of Frame

    # Before setting up splitt Block between this and next Block,
    # Check if Flag to skip creating split block is set (1=dont create split block)
    LOAD 4
    JUMPC > RETURN          # If no split block needed, done. Pointer already on arg stack

    # Now create split Block after this one
    LOADI 0
    STOREIN 1 1             # Mark as unused
    # Calculate split Block size
    LOAD 3                  # Original Block Size
    SUB 2                   # Sub Frame Size of new Frame
    STOREIN 1 0             # Store split block size
    STORE 2
    
    # Store split Block size at end of split Frame
    MOVE in1 acc
    ADD 2                   # Points to next Frame
    MOVE acc in1
    LOAD 2
    STOREIN 1 -1            # Store new Frame size at end of Frame
    JUMP RETURN
    # Done with this branch

REALLOC_NO_SHRINK:
    # Check if Frame is last one
    MOVE in1 acc
    SUB 4095
    JUMPC < REALLOC_NOT_END
    LOAD 2                  # Load wanted Size
    ADDI 3                  # Add header size
    STOREIN 1 0             # Store at Frame Start
    STORE 3
    MOVE in1 acc
    ADD 3
    MOVE acc in1            # Jump to end of frame
    LOAD 3                  # Load Frame Size
    STOREIN 1 -1          # Update end Frame size
    JUMP RETURN             # Return

REALLOC_NOT_END:
    # Check if next Frame is Free and big enough and Split is usefull
    LOADIN 1 0
    STORE 3
    MOVE in1 acc
    ADD 3
    MOVE acc in1
    LOADIN 1 1              # Load flag if used
    JUMPC > REALLOC_HOPELESS    # Jumpes if next Frame is used
    LOADIN 1 0
    STORE 4                     # Store Size of next Frame
    LOAD 2                  # Load wanted data size
    ADDI 3                  # Add header
    SUB 3                   # Minus (header+old data size) = minimum size of next Block
    SUB 4                   # Subtract actual size of next Block
    JUMPC > REALLOC_HOPELESS    # Zero or negativ if Space is enough
    # Check if Split of next Frame is usefull
    ADDI 4                  # Leftover Space is NEGATIVE, so if <= 0, split
    JUMPC <= REALLOC_SPLIT
    # Just put two Frames together. No split of next Frame.
    MOVE in1 acc
    SUB 3
    MOVE acc in1
    LOADIN 1 0
    ADD 4
    STOREIN 1 0
    STORE 3                 # Store new size of Frame. Old Size not longer needed 
    MOVE in1 acc
    ADD 3
    MOVE acc in1            # Point to next Frame
    LOAD 3
    STOREIN 1 -1            # Store Frame size at Frame end
    JUMP RETURN

    # Next block can be reused and split. Resize old Block, then do rest
REALLOC_SPLIT:
    MOVE in1 acc
    SUB 3
    MOVE acc in1
    LOAD 2                  # Wanted size
    ADDI 3
    STOREIN 1 0
    LOAD 3                  # Size of old Block
    ADD 4                   # Size of next Block (befor split)
    SUB 2
    SUBI 3                  # Overall size - Size of new Block = size of split Block
    STORE 4                 # Store new size of split Block at M(4)
    LOADIN 1 0
    STORE 3                 # Store new size of old Block in 3
    MOVE in1 acc
    ADD 3
    MOVE acc in1            # Point to end of old Block
    LOAD 3
    STOREIN 1 -1            # Set size at end of old Frame
    LOAD 4
    STOREIN 1 0             # Set size of new split Frame
    LOADI 0
    STOREIN 1 1             # Mark split Frame as unused
    MOVE in1 acc
    ADD 4
    MOVE acc in1            # Point to end of Split Frame
    LOAD 4
    STOREIN 1 -1            # Update size pointer at end of Split Frame
    JUMP RETURN             # Adress of Heap didnt change, so old pointer in arg is correct

REALLOC_HOPELESS:
    # Old Block couldnt be reused in any way. alloc new space, then copy.
    # Call Malloc
    LOADI 1
    STORE 1026
    LOAD 2                  # Size of new Data is at M(2)
    STORE 1030
    LOADI *MALLOC
    STORE 1024
    MOVE pc acc
    STORE 1025
    JUMP CALL

    LOAD 0                  # Load Pointer to return cell
    MOVE acc in1
    LOAD 1030               # Points at data-cells of new frame, not frame start
    STOREIN 1 0             # Store new Pointer at return cell
    STORE 4

    # Now copy Data from old frame to new Frame. New Frame > Old Frame!
    LOAD 1                  # Pointer to old Frame start
    ADDI 2                  # Point to old data start
    STORE 3                 
    MOVE acc in1
    LOADIN 1 -2             # Load size of old Frame
    SUBI 3
    STORE 2                 # Amount of cells to copy
REALLOC_COPY_LOOP:
    LOAD 2
    JUMPC <= REALLOC_COPY_DONE
    SUBI 1
    STORE 2                 # Decrease Loop Counter
    LOAD 3
    MOVE acc in1
    LOADIN 1 0
    STORE 5                 # Store Data in M(5)
    MOVE in1 acc
    ADDI 1                  # Increase Old data pointer by one
    STORE 3
    LOAD 4
    MOVE acc in1
    LOAD 5
    STOREIN 1 0             # Copy Data to new Frame
    MOVE in1 acc
    ADDI 1                  # Increase new data pointer by one
    STORE 4
    JUMP REALLOC_COPY_LOOP  

REALLOC_COPY_DONE:
    # Free old Block, then return. Return val is already stored
    LOADI 1
    STORE 1026
    LOAD 1                  # Points to beginn of Frame
    ADDI 2                  # Points to beginn of data as Free(n) expects
    STORE 1030
    LOADI *FREE
    STORE 1024
    MOVE pc acc
    STORE 1025
    JUMP CALL
    JUMP RETURN

###### CALLOC ######
CALLOC:
    # Call Malloc
    LOADIN 1 0          # load n
    STORE 1030          # store as argument
    LOADI 1
    STORE 1026          # set arg count to 1
    LOADI *MALLOC       # Offset to MALLOC
    STORE 1024
    MOVE pc acc
    STORE 1025
    JUMP CALL           # CALL

    LOAD 1030           # Pointer to data-start
    STOREIN 1 0
    STORE 2
    MOVE acc in1
    LOADIN 1 -2         # Load frame size
    SUBI 3              # subtract header size
    STORE 3             # M(3) = Loop Counter
    LOADI 0
    STORE 4

    CALLOC_LOOP:
    LOAD 3
    JUMPC <= CALLOC_DONE    # if <= done with filling zero
    SUBI 1
    STORE 3             # decrease counter

    LOAD 4              # Load 0
    STOREIN 1 0         # Copy 0 to data-cell

    LOAD 2              # Load pointer
    ADDI 1              # increase pointer
    STORE 2             # store new pointer
    MOVE acc in1        # put pointer into in1
    JUMP CALLOC_LOOP

CALLOC_DONE:
    # Der Daten-Pointer steht noch in M(1030) (von MALLOC geschrieben).
    # Damit ist der Rückgabewert bereits an der richtigen Stelle.
    JUMP RETURN
