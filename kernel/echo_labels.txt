ECHO_ENTRY:
    LOADI 32
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *MALLOC
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    STORE 0

ECHO_LOOP:
    LOADI 0
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *READ_KEY
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOAD 1030
    STORE 2
    STORE 1030
    LOAD 0
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *ITOA
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOAD 0
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *PRINT
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    PRINTC '\n'

    LOAD 2
    SUBI 27
    JUMPC != ECHO_LOOP

    PRINTC "Shutdown!\n"
    LOADI 0
    STORE 1026
    LOADI *SHUTDOWN
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP SHUTDOWN
