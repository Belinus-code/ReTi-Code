###### CALL METHOD ######
# 1. Save IN1 to Stack
MOVE in1 acc
STOREIN 2 0             # Dont Increse Stack now to save code

# 2. Save M(0) to M(4) to stack
LOAD 0
STOREIN 2 1             # Manuell Increment of Stack to save Code
LOAD 1
STOREIN 2 2
LOAD 2
STOREIN 2 3
LOAD 3
STOREIN 2 4
LOAD 4
STOREIN 2 5             # Stack +6

# 3. Copy n Arguments to stack
LOADI 0
STORE 1027              # Use 1027 as loop counter (i)
LOAD 1026               # Load arg count (n)
SUB 1027
JUMPC <= 13             # If i >= n, continue
LOADI 1030
ADD 1027                # Address: 1030 + i
MOVE acc in1
LOADIN 1 0              # Load argument from transfer area
STOREIN 2 6             # Push to stack
MOVE in2 acc            # Increment stack pointer
ADDI 1
MOVE acc in2
LOAD 1027               # Increment loop counter
ADDI 1
STORE 1027
JUMP -14

# 4. Save Amount of Args (n) to Stack
LOAD 1026               # M (1026) is arg count
STOREIN 2 6

# 5. Save Absolute Return Address to Stack (PC_ref + 3)
LOAD 1025
ADDI 3                  # Copying PC to M(1025) must be third last instruction
STOREIN 2 7
MOVE in2 acc
ADDI 8                  # finally update Stack-Pointer
MOVE acc in2

# 6. Save Adress of first Argument to M(0), just for QOL
SUBI 2                  # Stack-Counter - 2 Points to arg_count
SUB 1026                # Now points to first Argument
MOVE acc in1
STORE 0

# 7. Jump to Absolute Method Address (Offset + PC_ref - 2)
LOAD 1025
ADD 1024
SUBI 2
MOVE acc pc
