###### CALL METHOD ######
    # 1. Save IN1 to Stack
    MOVE IN1 acc

    STOREIN 2 9

    STOREIN 2 10             # Stack +11

    LOADI 0
    STORE 1027              # Use 1027 as loop counter (i)
    LOAD 1026               # Load arg count (n)
    SUB 1027
    JUMPC <= 13             # If i >= n, continue
    LOADI 1030
    ADD 1027                # Address: 1030 + i
    MOVE acc IN1
    LOADIN 1 0              # Load argument from transfer area
    STOREIN 2 11             # Push to stack
    MOVE in2 acc            # Increment stack pointer
    ADDI 1
    MOVE acc in2
    LOAD 1027               # Increment loop counter
    ADDI 1
    STORE 1027
    JUMP -14               

# 4. Save Amount of Args (n) to Stack
    LOAD 1026               # M (1026) is arg count
    STOREIN 2 11

    # 5. Save Absolute Return Address to Stack (PC_ref + 3)
    LOAD 1025
    ADDI 3                  # Copying PC to M(1025) must be third last instruction
    STOREIN 2 12
    MOVE in2 acc
    ADDI 13                  # finally update Stack-Pointer
    MOVE acc in2

    # 6. Save Adress of first Argument to M(0) and IN1, just for QOL

    SUB 1026                # Now points to first Argument
    MOVE acc IN1
    STORE 0

    # 7. Jump to Absolute Method Address (Offset + PC_ref - 2)
    LOAD 1025
    ADD 1024
    SUBI 2
    MOVE acc pc
