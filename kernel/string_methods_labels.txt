##### MERGE_HEAP_STRINGS #####
MERGE_HEAP_STRINGS:
    LOADIN 1 0
    STORE 1
    LOADIN 1 1
    STORE 2
    SUB 1
    JUMPC = EXCEPTION       # In Case both Strings are the same
    LOAD 1
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *GET_STRING_SIZE
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    STORE 3

    LOAD 2
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *GET_STRING_SIZE
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    ADD 3
    ADDI 1              # For Zero Terminator
    STORE 3             # M(3) holds needed size

    LOAD 1
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *SIZEOF
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    SUB 3
    JUMPC >= MERGE_HEAP_STRINGS_NOMAL
    # Need to realloc Main Buffer
    LOAD 1
    STORE 1030
    LOAD 3
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *REALLOC
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030
    STORE 1
MERGE_HEAP_STRINGS_NOMAL:
    LOAD 1
    STORE 1030
    LOAD 2
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *STRCAT
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    # Now Free secondary Buffer
    LOAD 2
    STORE 1030
    LOADI 1
    STORE 1026
    LOADI *FREE
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 0
    MOVE ACC IN1
    LOAD 1
    STOREIN 1 0
    JUMP RETURN

##### PRINT #####
PRINT:
    LOADIN 1 0
    MOVE ACC IN1
PRINT_LOOP:
    LOADIN 1 0
    JUMPC <= RETURN
    STORE 1
    PRINTC 1
    MOVE IN1 ACC
    ADDI 1
    MOVE ACC IN1
    JUMP PRINT_LOOP

###### STRCAT ######
STRCAT:
    LOADIN 1 0
    STORE 1             # Pointer to Main String
    LOADIN 1 1
    STORE 2             # Pointer to Secondary String
    LOAD 1
    MOVE ACC IN1
SRTCAT_LOOP1:
    LOADIN 1 0
    JUMPC = STRCAT_FIRST_ZERO
    MOVE IN1 ACC
    ADDI 1
    MOVE ACC IN1
    JUMP SRTCAT_LOOP1
STRCAT_FIRST_ZERO:
    MOVE IN1 ACC
    STORE 1
STRCAT_LOOP2:
    LOAD 2
    MOVE ACC IN1
    LOADIN 1 0
    JUMPC = RETURN
    STORE 3
    LOAD 1
    MOVE ACC IN1
    LOAD 3
    STOREIN 1 0
    MOVE IN1 ACC
    ADDI 1
    STORE 1
    JUMP STRCAT_LOOP2

###### GET_STRING_SIZE ######
GET_STRING_SIZE:
    LOADI 0
    STORE 3                 # Counter for Size
    STORE 4                 # Flag if Max-size if given
    LOADIN 1 0
    STORE 1
    LOADIN 2 -2             # Load Arg Count
    SUBI 1
    JUMPC = GET_STRING_SIZE_NO_LIMIT
    LOADI 1
    STORE 4                 # Set Flag
    LOADIN 1 1
    ADD 1
    SUBI 1
    STORE 2                 # Last Cell within String
    LOAD 1
    MOVE ACC IN1

GET_STRING_SIZE_LOOP1:
    LOADIN 1 0
    JUMPC = GET_STRING_SIZE_DONE
    MOVE IN1 ACC
    ADDI 1
    MOVE ACC IN1
    STORE 5
    LOAD 4
    JUMPC = GET_STRING_SIZE_SKIP_LIMIT_CHECK
    LOAD 5
    SUB 2
    JUMPC > GET_STRING_SIZE_ERR
GET_STRING_SIZE_SKIP_LIMIT_CHECK:
    LOAD 3
    ADDI 1
    STORE 3
    JUMP GET_STRING_SIZE_LOOP1

GET_STRING_SIZE_NO_LIMIT:
    LOAD 1
    MOVE ACC IN1
GET_STRING_SIZE_LOOP2:
    LOADIN 1 0
    JUMPC = GET_STRING_SIZE_DONE
    MOVE IN1 ACC
    ADDI 1
    MOVE ACC IN1
    LOAD 3
    ADDI 1
    STORE 3
    JUMP GET_STRING_SIZE_LOOP2

GET_STRING_SIZE_DONE:
    LOAD 0
    MOVE ACC IN1
    LOAD 3
    STOREIN 1 0
    LOAD 4
    JUMPC = RETURN
    LOADI 1
    STOREIN 1 1
    JUMP RETURN

GET_STRING_SIZE_ERR:
    LOAD 0
    MOVE ACC IN1
    LOADIN 1 1
    STOREIN 1 0         # Copy Max Size as return arg
    LOADI 0
    STOREIN 1 1
    JUMP RETURN

###### ATOI METHOD ######
ATOI:
    LOADI 0
    STORE 4
    STORE 5
    LOADIN 1 0              # Pointer to String
    STORE 1
    LOADIN 2 -2
    STORE 2
    SUBI 1                  # If only one argument, then skip following
    JUMPC = ATOI_NO_OFFSET
    LOADIN 1 1              # Load offset
    STORE 3
    LOAD 1
    ADD 3
    STORE 1

ATOI_NO_OFFSET:
    LOAD 1
    MOVE ACC IN1
    LOADIN 1 0
    SUB 145                 # Ascii '-' (negative Value)
    STORE 3                 # M(3) is zero if first char is negative, != 0 if not
    JUMPC != ATOI_PARSE_LOOP
    LOAD 1
    ADDI 1
    STORE 1

ATOI_PARSE_LOOP:
    LOAD 1
    MOVE ACC IN1
    LOADIN 1 0
    SUBI 48
    STORE 2
    JUMPC < ATOI_END_FOUND
    SUBI 9
    JUMPC > ATOI_END_FOUND
    # Call multiply method and multiply old buffer by ten, then add number to buffer
    LOADI 2                 # 2 Arguments
    STORE 1026
    LOAD 4                  # Load Result of parser
    STORE 1030              # Parse result to args
    LOADI 10                # -1 to negate
    STORE 1031              # store -1 to args
    LOADI *MUL
    STORE 1024
    MOVE pc acc
    STORE 1025
    JUMP CALL
    LOAD 1030
    ADD 2
    STORE 4
    LOAD 1
    ADDI 1
    STORE 1
    JUMP ATOI_PARSE_LOOP


ATOI_END_FOUND:
    LOAD 3
    JUMPC != ATOI_END       # Jump if no - found before number
    LOADI 2                 # 2 Arguments
    STORE 1026
    LOAD 4                  # Load Result of parser
    STORE 1030              # Parse result to args
    LOADI -1                # -1 to negate
    STORE 1031              # store -1 to args
    LOADI *MUL
    STORE 1024
    MOVE pc acc
    STORE 1025
    JUMP CALL
    LOAD 1030
    STORE 4

ATOI_END:
    LOAD 0
    MOVE ACC IN1
    LOADIN 2 -2
    SUBI 1
    JUMPC = ATOI_NO_OFFSET_RET
    LOADIN 1 0
    STORE 2
    LOAD 1
    SUB 2
    STOREIN 1 1
ATOI_NO_OFFSET_RET:
    LOAD 4
    STOREIN 1 0
    JUMP RETURN

###### ITOA METHOD ######
ITOA:
    LOADIN 1 0              # Integer to Convert
    STORE 1
    LOADIN 1 1              # String Buffer (expected to be big enough)
    STORE 2
    MOVE IN2 ACC
    ADDI 11
    MOVE ACC IN2            # Update Stack Pointer 

    LOADI 1000000000
    STOREIN 2 -11
    LOADI 100000000
    STOREIN 2 -10
    LOADI 10000000
    STOREIN 2 -9
    LOADI 1000000
    STOREIN 2 -8
    LOADI 100000
    STOREIN 2 -7
    LOADI 10000
    STOREIN 2 -6
    LOADI 1000
    STOREIN 2 -5
    LOADI 100
    STOREIN 2 -4
    LOADI 10
    STOREIN 2 -3
    LOADI 1
    STOREIN 2 -2
    LOADI 0
    STOREIN 2 -1

    LOADI 0
    STORE 3                 # Flag if Zeros have to be printed
    LOAD 2
    MOVE ACC IN1
    LOAD 1
    JUMPC >= ITOA_POS
    LOAD '-'
    STOREIN 1 0
    MOVE IN1 ACC
    ADDI 1
    MOVE ACC IN1
    LOADI 0
    SUB 1
    STORE 1                 # Store Value of negative Number in 1

ITOA_POS:
    MOVE IN2 ACC
    SUBI 11                 # Points to 1.000.000.000
    STORE 6                 # Holds Pointer
    LOADIN 2 -11
    STORE 4
ITOA_LOOP:
    LOAD 1
    STORE 1030
    LOAD 4
    STORE 1031
    LOADI 2
    STORE 1026
    LOADI *DIVMOD
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL
    LOAD 1030               # quotient
    STORE 5
    JUMPC != ITOA_LOOP_PRINT
    LOAD 3
    JUMPC = ITOA_LOOP_CONT
ITOA_LOOP_PRINT:
    LOAD '0'
    ADD 5
    STOREIN 1 0
    MOVE IN1 ACC
    ADDI 1
    MOVE ACC IN1
    LOADI 1
    STORE 3
ITOA_LOOP_CONT:
    LOAD 1031               # remainder
    STORE 1                 # update remainder
    MOVE IN1 ACC
    STORE 7
    LOAD 6
    ADDI 1
    STORE 6
    MOVE ACC IN1
    LOADIN 1 0
    STORE 4
    JUMPC = ITOA_DONE
    LOAD 7
    MOVE ACC IN1
    JUMP ITOA_LOOP

ITOA_DONE:
    LOAD 7
    MOVE ACC IN1
    LOAD 3
    JUMPC != ITAO_NOT_ZERO
    LOADI '0'
    STOREIN 1 0
    MOVE IN1 ACC
    ADDI 1
    MOVE ACC IN1

ITAO_NOT_ZERO:
    LOADI 0
    STOREIN 1 0             # Zero Terminator
    MOVE IN2 ACC
    SUBI 11
    MOVE ACC IN2            # Update Stack Pointer 
    JUMP RETURN
