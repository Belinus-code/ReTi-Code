##### GET LINE WIDTH METHOD #####
# Get the Width of the current Line, thus the amount of visable characters.
GET_LINE_WIDTH:
LOADIN 1 0
STORE 1             # Start of String
LOADIN 1 1
STORE 2             # End of String
LOADIN 1 2
STORE 3             # Pointer to a char in specific line
STORE 6             # Target Character relative Position
LOADI 0            
STORE 5
LOADI 1
STORE 4

# Boundry Checks (is cursor in String)
LOAD 3
SUB 1
JUMPC < GET_LINE_WIDTH_ERROR
LOAD 2
SUB 3
JUMPC < GET_LINE_WIDTH_ERROR

# If Target Character is a newline, there is more logic to do
LOAD 3
MOVE ACC IN1
LOADIN 1 0
SUB 110
JUMPC != GET_LINE_WIDTH_FIND_START
# Target Char is newline, check if target char is first char of string:
LOAD 3
SUB 1
JUMPC = GET_LINE_WIDTH_LOOP2S
LOAD 3
SUBI 1
STORE 3


# Find beginning of Line
GET_LINE_WIDTH_FIND_START:
LOAD 3
MOVE ACC IN1
GET_LINE_WIDTH_LOOP1:
LOADIN 1 0
SUB 110
JUMPC = GET_LINE_WIDTH_LOOP1E2
MOVE IN1 ACC
SUB 1
JUMPC <= GET_LINE_WIDTH_LOOP1E1
MOVE IN1 ACC
SUBI 1
MOVE ACC IN1
JUMP GET_LINE_WIDTH_LOOP1

GET_LINE_WIDTH_LOOP1E1:
MOVE IN1 ACC
STORE 3
JUMP GET_LINE_WIDTH_LOOP2S

GET_LINE_WIDTH_LOOP1E2:
MOVE IN1 ACC
ADDI 1              # IN1 points to newline, add one to point to first char
MOVE ACC IN1
STORE 3
LOADI 0
STORE 4             # Clear Flag that Line is first line of String

GET_LINE_WIDTH_LOOP2S:
LOAD 6
SUB 3
ADDI 1
STORE 6

GET_LINE_WIDTH_LOOP2:
MOVE IN1 ACC
SUB 2
JUMPC >= GET_LINE_WIDTH_LOOP2E1
LOADIN 1 0
SUB 110
JUMPC = GET_LINE_WIDTH_LOOP2E2
MOVE IN1 ACC
ADDI 1
MOVE ACC IN1
JUMP GET_LINE_WIDTH_LOOP2

GET_LINE_WIDTH_LOOP2E1:
LOADI 1
STORE 5
JUMP GET_LINE_WIDTH_END

GET_LINE_WIDTH_LOOP2E2:
MOVE IN1 ACC
SUBI 1
MOVE ACC IN1

GET_LINE_WIDTH_END:
MOVE IN1 ACC
SUB 3
ADDI 1
JUMPC >= 2
LOADI 0                 # If width is negativ, set it to zero
STORE 3
LOAD 0
MOVE ACC IN1
LOAD 3
STOREIN 1 0
LOAD 6
STOREIN 1 1
LOAD 4
STOREIN 1 2
LOAD 5
STOREIN 1 3
JUMP RETURN

# In an Event of an Error, return -1
GET_LINE_WIDTH_ERROR:
LOAD 0
MOVE ACC IN1
LOADI -1
STOREIN 1 0
JUMP RETURN

                    # As a non interrupted String

STORE 4

STORE 5

STORE 7             # Target Line Counter

STORE 7             # Target Line Counter

JUMPC != 4
LOAD 6
ADDI 1
STORE 6
LOAD 3
SUB 1
JUMPC != 3
LOAD 6
STORE 7             # If Char is target char, copy line count

ADDI 1
STORE 1

JUMPC > GET_LINE_COUNT_DONE

SUB 4               # Check if Pointer is on Gap Start
JUMPC != GET_LINE_COUNT_LOOP

ADDI 1
STORE 10

STORE 1

SUB 2
JUMPC > RETURN

ADDI 1
MOVE ACC IN1

JUMPC != PRINT_LINE_FIND_LOOP

    STORE 1024

    STORE 1025

    STORE 1026
    LOADI *SIZEOF
    STORE 1024
    MOVE PC ACC
    STORE 1025
    JUMP CALL

    LOAD 1
